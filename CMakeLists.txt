cmake_minimum_required(VERSION 3.21)
project(RAMD_Docs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
find_path(LOK_INCLUDE_DIR LibreOfficeKit/LibreOfficeKitGtk.h REQUIRED)
find_library(LOK_GTK_LIBRARY NAMES libreofficekitgtk REQUIRED)
add_executable(ramd-docs src/main.cpp)
target_include_directories(ramd-docs PRIVATE ${LOK_INCLUDE_DIR})
target_compile_definitions(ramd-docs PRIVATE LOK_USE_UNSTABLE_API)
target_link_libraries(ramd-docs PRIVATE PkgConfig::GTK PkgConfig::JSON ${LOK_GTK_LIBRARY})
target_compile_options(ramd-docs PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS ramd-docs RUNTIME DESTINATION bin)
install(FILES packaging/ramd-docs.desktop DESTINATION share/applications)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ramd-docs)
set(CPACK_PACKAGE_CONTACT "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libreoffice-writer, libreoffice-gtk3, liblibreofficekitgtk")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
include(CTest)
if(BUILD_TESTING)
  add_executable(engine-check tests/engine_check.cpp)
  target_include_directories(engine-check PRIVATE ${LOK_INCLUDE_DIR})
  target_link_libraries(engine-check PRIVATE ${CMAKE_DL_LIBS})
endif()
