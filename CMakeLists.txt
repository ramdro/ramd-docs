cmake_minimum_required(VERSION 3.21)
project(RAMD_Docs VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
if(WIN32)
  set(LOK_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/windows-lok/include")
  add_library(lokgtk STATIC windows-lok/lokdocview.cxx windows-lok/tilebuffer.cxx)
  target_include_directories(lokgtk PRIVATE "${LOK_INCLUDE_DIR}" windows-lok)
  target_compile_definitions(lokgtk PRIVATE LOK_USE_UNSTABLE_API LOK_DOC_VIEW_IMPLEMENTATION LOK_PATH="C:/Program Files/LibreOffice/program")
  target_link_libraries(lokgtk PRIVATE PkgConfig::GTK)
  set(LOK_GTK_LIBRARY lokgtk)
else()
  find_path(LOK_INCLUDE_DIR LibreOfficeKit/LibreOfficeKitGtk.h REQUIRED)
  find_library(LOK_GTK_LIBRARY NAMES libreofficekitgtk REQUIRED)
endif()
add_executable(ramd-docs src/main.cpp)
target_include_directories(ramd-docs PRIVATE ${LOK_INCLUDE_DIR})
target_compile_definitions(ramd-docs PRIVATE LOK_USE_UNSTABLE_API)
target_link_libraries(ramd-docs PRIVATE PkgConfig::GTK PkgConfig::JSON ${LOK_GTK_LIBRARY})
target_compile_options(ramd-docs PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS ramd-docs RUNTIME DESTINATION bin)
install(FILES packaging/ramd-docs.desktop DESTINATION share/applications)
install(FILES packaging/ramd-docs.svg DESTINATION share/icons/hicolor/scalable/apps)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ramd-docs)
set(CPACK_PACKAGE_CONTACT "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libreoffice-writer, libreoffice-gtk3, liblibreofficekitgtk, adwaita-icon-theme, librsvg2-common")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "Proprietary")
set(CPACK_RPM_PACKAGE_REQUIRES "libreoffice-writer, libreoffice-gtk3, libreofficekit, adwaita-icon-theme, librsvg2")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/usr/share/icons" "/usr/share/icons/hicolor" "/usr/share/icons/hicolor/scalable" "/usr/share/icons/hicolor/scalable/apps" "/usr/share/applications")
include(CPack)
include(CTest)
if(BUILD_TESTING)
  add_executable(ui-check tests/ui_check.cpp)
  target_compile_definitions(ui-check PRIVATE LOK_USE_UNSTABLE_API)
  target_include_directories(ui-check PRIVATE ${LOK_INCLUDE_DIR})
  target_link_libraries(ui-check PRIVATE PkgConfig::GTK PkgConfig::JSON ${LOK_GTK_LIBRARY})
  add_executable(engine-check tests/engine_check.cpp)
  target_include_directories(engine-check PRIVATE ${LOK_INCLUDE_DIR})
  target_link_libraries(engine-check PRIVATE ${CMAKE_DL_LIBS})
endif()
